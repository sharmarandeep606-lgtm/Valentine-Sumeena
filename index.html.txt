  <script>
    const yesBtn = document.getElementById("yesBtn");
    const noBtn = document.getElementById("noBtn");
    const result = document.getElementById("result");
    const resultTitle = document.getElementById("resultTitle");
    const resultText = document.getElementById("resultText");
    const musicBtn = document.getElementById("musicBtn");
    const bgm = document.getElementById("bgm");

    const noMessages = [
      "Please don’t break my heart 😭💔 Press YES 🥺",
      "Stop playing 🙈 I love you 💖 Press YES",
      "Pretty pleaseeee 😭👉👈",
      "You know you love me 😘 Press YES",
      "Unlimited hugs if you press YES 🤗",
      "Okay last chance 😤 (not really)"
    ];

    let noCount = 0;
    let yesScale = 1;

    musicBtn.onclick = async () => {
      try {
        await bgm.play();
        musicBtn.textContent = "⏸ Pause Music";
      } catch (e) {}
    };

    yesBtn.onclick = () => {
      resultTitle.textContent = "Congratulations 💘";
      resultText.textContent =
        "You are officially my Valentine, Sumeena 💖 I love you so much 🌹";
      result.classList.add("show");
      yesBtn.disabled = true;
      noBtn.disabled = true;
    };

    noBtn.onclick = () => {
      noCount++;
      yesScale = Math.min(1.8, 1 + noCount * 0.15);
      yesBtn.style.transform = `scale(${yesScale})`;

      resultTitle.textContent = "Noooo 😭";
      resultText.textContent =
        noMessages[(noCount - 1) % noMessages.length];
      result.classList.add("show");
    };
  </script>
